# Two dependencies, deliberately. This runs on a Windows box that somebody else
# has to keep working, and every extra install is a thing that can go missing.
#
# pymssql rather than pyodbc: it bundles FreeTDS in the wheel and needs no
# unixODBC or msodbcsql driver installed on the host.
pymssql==2.4.1

# psycopg2-binary rather than psycopg2: no build toolchain needed on Windows.
# 2.9.11 and not 2.9.10: 2.9.10's wheels stop at CPython 3.13, so on a box with
# 3.14 pip falls back to building from source and fails for want of a compiler --
# which is exactly the toolchain this line exists to avoid needing.
psycopg2-binary==2.9.11
